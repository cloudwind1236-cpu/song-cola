1. database.rules.json （強制 1金+4紅、5位選手絕對不重複、一人限投一次） 
{
  "rules": {
    "config": {
      ".read": true,
      ".write": "auth != null && auth.token.email === 'host@yourcontest.com'"
    },
    "candidates": {
      ".read": true,
      "$cid": {
        "cc": {
          // 僅允許在開放投票期間累加 +100cc (紅罐) 或 +200cc (黃金罐)
          ".write": "auth != null && root.child('config/phase').val() === 'VOTING_OPEN'",
          ".validate": "newData.val() === data.val() + 100 || newData.val() === data.val() + 200"
        },
        "goldCount": {
          // 同分破局指標：每次僅能 +1
          ".write": "auth != null && root.child('config/phase').val() === 'VOTING_OPEN'",
          ".validate": "newData.val() === data.val() + 1"
        },
        "redCount": {
          ".write": "auth != null && root.child('config/phase').val() === 'VOTING_OPEN'",
          ".validate": "newData.val() === data.val() + 1"
        }
      }
    },
    "votes": {
      "$lineUid": {
        ".read": "auth != null",
        // 1. 必須在開放投票階段 2. 該 LINE UID 先前未曾投過票 (!data.exists())
        ".write": "auth != null && root.child('config/phase').val() === 'VOTING_OPEN' && !data.exists()",
        "picks": {
          // 驗證必含 1 張金票 (gold) 與 4 張依字母排序之紅票 (r1 < r2 < r3 < r4)
          // 且 gold 絕對不等於 r1~r4 任一選手，100% 保證投滿 5 位不同選手！
          ".validate": "newData.hasChildren(['gold', 'r1', 'r2', 'r3', 'r4']) &&
            newData.child('r1').val() < newData.child('r2').val() &&
            newData.child('r2').val() < newData.child('r3').val() &&
            newData.child('r3').val() < newData.child('r4').val() &&
            newData.child('gold').val() !== newData.child('r1').val() &&
            newData.child('gold').val() !== newData.child('r2').val() &&
            newData.child('gold').val() !== newData.child('r3').val() &&
            newData.child('gold').val() !== newData.child('r4').val()"
        }
      }
    }
  }
}
2. liff-cola-vote.js （LINE LIFF 自動讀取 userId ＋ 1金4紅原子化送票程式碼） 
import liff from '@line/liff';
import { getDatabase, ref, update, increment, serverTimestamp } from 'firebase/database';

const db = getDatabase();

// 1. 觀眾用 LINE 掃描全場共用 QR 碼後，自動無感取得唯一 LINE userId
async function initLineLiff() {
  await liff.init({ liffId: 'YOUR_LINE_LIFF_ID' });
  if (!liff.isLoggedIn()) {
    liff.login();
    return;
  }
  const profile = await liff.getProfile();
  window.currentLineUser = {
    uid: profile.userId,         // 唯一識別碼，無痕視窗也無法偽造
    name: profile.displayName,   // 觀眾 LINE 暱稱
    avatar: profile.pictureUrl
  };
}

// 2. 一次性送出「1罐黃金可樂(+200cc) + 4罐經典紅罐(+100cc)」
async function submitColaBallot(lineUid, goldCandidateId, fourRedCandidateIds) {
  if (!goldCandidateId || fourRedCandidateIds.length !== 4) return;

  // 將 4 個紅罐選手編號依字母遞增排序，以通過 Firebase Security Rules 驗證
  const sortedReds = [...fourRedCandidateIds].sort();
  const updates = {};

  // 寫入該 LINE 觀眾的選票明細
  updates[`votes/${lineUid}`] = {
    picks: {
      gold: goldCandidateId,
      r1: sortedReds[0],
      r2: sortedReds[1],
      r3: sortedReds[2],
      r4: sortedReds[3]
    },
    timestamp: serverTimestamp()
  };

  // 為本命黃金罐選手原子累加 +200cc 及 1 罐金罐紀錄
  updates[`candidates/${goldCandidateId}/cc`] = increment(200);
  updates[`candidates/${goldCandidateId}/goldCount`] = increment(1);

  // 為 4 位經典紅罐選手各原子累加 +100cc 及 1 罐紅罐紀錄
  sortedReds.forEach((cid) => {
    updates[`candidates/${cid}/cc`] = increment(100);
    updates[`candidates/${cid}/redCount`] = increment(1);
  });

  // 單次請求原子化寫入，100人同時按下也絕不掉票！
  await update(ref(db), updates);
}
